## 🔧 Serviços e Hooks

### Serviços (Singleton)

```typescript
// ✅ BOM: Serviço singleton com responsabilidade única
class GlobalCacheService {
  private static instance: GlobalCacheService
  private cache: Map<string, any> = new Map()
  
  static getInstance(): GlobalCacheService {
    if (!this.instance) {
      this.instance = new GlobalCacheService()
    }
    return this.instance
  }
  
  async get(key: string): Promise<any> {
    // ...
  }
}

export const globalCacheService = GlobalCacheService.getInstance()
```

### Hooks (Stateful)

```typescript
// ✅ BOM: Hook com responsabilidade clara
export function useSuppliers() {
  const [data, setData] = useState<Supplier[] | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)
  
  useEffect(() => {
    globalCacheService.getSuppliers()
      .then(setData)
      .catch(err => setError(err.message))
      .finally(() => setLoading(false))
  }, [])
  
  return { data, loading, error }
}
```

### Estrutura de Componente

```typescript
  title: string
  onAction: () => void
}

export function MyComponent({ title, onAction }: MyComponentProps) {
  // 1. Hooks
  const [state, setState] = useState(false)
  const { data, loading } = useData()
  
  // 2. Callbacks
  const handleClick = useCallback(() => {
    setState(true)
    onAction()
  }, [onAction])
  
  // 3. Effects
  useEffect(() => {
    // ...
  }, [])
  
  // 4. Early returns
  if (loading) return <Loading />
  if (!data) return <Empty />
  
  // 5. Render
  return (
    <div>
      <h1>{title}</h1>
      <button onClick={handleClick}>Action</button>
    </div>
  )
}
```

## 🚫 Anti-Padrões a Evitar

### 1. Código Duplicado

```typescript
// ❌ RUIM: Código duplicado
function loadSuppliers() {
  const manifest = await fetchManifest()
  const data = await fetchSuppliers(manifest)
  return data
}

function loadDeputies() {
  const manifest = await fetchManifest()
  const data = await fetchDeputies(manifest)
  return data
}

// ✅ BOM: Abstração reutilizável
async function loadCache<T>(
  fetcher: (manifest: Manifest) => Promise<T>
): Promise<T> {
  const manifest = await fetchManifest()
  return fetcher(manifest)
}

const suppliers = await loadCache(fetchSuppliers)
const deputies = await loadCache(fetchDeputies)
```

### 2. Magic Numbers/Strings

```typescript
// ❌ RUIM: Magic numbers
if (items.length > 100) {
  paginate(items, 50)
}

// ✅ BOM: Constantes nomeadas
const MAX_ITEMS_BEFORE_PAGINATION = 100
const ITEMS_PER_PAGE = 50

if (items.length > MAX_ITEMS_BEFORE_PAGINATION) {
  paginate(items, ITEMS_PER_PAGE)
}
```

### 3. Funções Gigantes

```typescript
// ❌ RUIM: Função faz tudo
function processData(data: any) {
  // 200 linhas de código...
}

// ✅ BOM: Funções pequenas e focadas
function processData(data: RawData): ProcessedData {
  const validated = validateData(data)
  const normalized = normalizeData(validated)
  const enriched = enrichData(normalized)
  return enriched
}

function validateData(data: RawData): ValidatedData { }
function normalizeData(data: ValidatedData): NormalizedData { }
function enrichData(data: NormalizedData): ProcessedData { }
```
